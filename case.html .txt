  <div id="root" class="card">載入中...</div>

  <script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
  <script>
    const SUPABASE_URL = 'https://YOUR_SUPABASE_PROJECT_ID.supabase.co'; 
    const SUPABASE_ANON_KEY = 'YOUR_SUPABASE_ANON_KEY';
    const sb = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    function esc(str) {
      if (!str) return '';
      return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
    }

    async function run() {
      const root = document.getElementById('root');
      try {
        const urlParams = new URLSearchParams(window.location.search);
        const caseId = urlParams.get('id');

        if (!caseId) {
          root.innerHTML = '<h2>網址缺乏案件編號</h2>';
          return;
        }

        const { data, error } = await sb
          .from('funeral_cases')
          .select('*')
          .eq('id', caseId)
          .maybeSingle();

        if (error || !data) {
          console.error('Database Error:', error);
          root.innerHTML = '<h2>找不到該案件資訊</h2>';
          return;
        }

        let schedule = data.schedule;
        if (typeof schedule === 'string') {
          try { schedule = JSON.parse(schedule); } catch(e) { schedule = []; }
        }
        if (!Array.isArray(schedule)) schedule = [];

        let mediaPaths = data.media_paths;
        if (typeof mediaPaths === 'string') {
          try { mediaPaths = JSON.parse(mediaPaths); } catch(e) { mediaPaths = []; }
        }
        if (!Array.isArray(mediaPaths)) mediaPaths = [];

        let html = `<h1>${esc(data.deceased_name || '案件詳情')}</h1>`;
        html += `<div class="date">${esc(data.service_date || '')}</div>`;
        html += `<div class="info" style="margin-top:15px">`;
        if (data.venue) html += `<div><b>地點：</b>${esc(data.venue)}</div>`;
        if (data.address) html += `<div><b>地址：</b>${esc(data.address)}</div>`;
        if (data.contact) html += `<div><b>聯絡：</b><a href="tel:${esc(data.contact)}">${esc(data.contact)}</a></div>`;
        html += `</div>`;

        html += `<h2 style="margin-top:25px">流程</h2><div class="timeline">`;
        if (schedule.length > 0) {
          html += schedule.map(x => `
            <div class="item">
              <div class="time">${esc(x.time || '')}</div>
              <div>
                <div class="type">${esc(x.type || '')}</div>
                ${x.people ? `<div class="people">${esc(x.people)}</div>` : ''}
                ${x.detail ? `<div class="detail">${esc(x.detail)}</div>` : ''}
              </div>
            </div>
          `).join('');
        } else {
          html += `<div class="muted">目前尚未建立流程。</div>`;
        }
        html += `</div>`;

        if (data.note) {
          html += `<h2>備註</h2><div style="white-space:pre-wrap;line-height:1.7">${esc(data.note)}</div>`;
        }

        if (data.program_path || mediaPaths.length > 0) {
          html += `<h2 style="margin-top:25px">檔案</h2>`;
        }

        if (data.program_path) {
          const { data: u } = sb.storage.from('funeral-files').getPublicUrl(data.program_path);
          html += `<a class="file" href="${u.publicUrl}" target="_blank">📄 開啟禮拜程序：${esc(data.program_name || 'PDF')}</a>`;
        }

        for (const f of mediaPaths) {
          if (!f || !f.path) continue;
          const { data: u } = sb.storage.from('funeral-files').getPublicUrl(f.path);
          const icon = (f.type && f.type.startsWith('video/')) ? '🎞️' : '🖼️';
          html += `<a class="file" href="${u.publicUrl}" target="_blank">${icon} ${esc(f.name || '檢視檔案')}</a>`;
        }

        root.innerHTML = html;
      } catch (err) {
        console.error('Unhandled Error:', err);
        root.innerHTML = '<h2>資料載入發生錯誤</h2>';
      }
    }

    run();
  </script>
